# LLM MANAGEMENT SYSTEM — Elasticsearch query demonstration
# Paste these requests into Kibana > Dev Tools > Console.
# Put the cursor inside one request and click the play button to run it.
# The seeded demo accounts use workspace_id "demo". Do not use these filters
# to inspect another user's private workspace.

# 1. Show the six Elasticsearch indices used by the application.
GET _cat/indices/users,projects,datasets,experiments,models,training_runs?v=true&h=index,health

# 2. Show the actual mapping (schema) for a training run.
# Notice keyword IDs/status, numeric metrics, dates, objects, and nested arrays.
GET training_runs/_mapping

# 3. Count the sample datasets in the demo workspace.
GET datasets/_count
{
  "query": { "term": { "workspace_id": "demo" } }
}

# 4. Read a few sample dataset documents without exposing user credentials.
GET datasets/_search
{
  "size": 5,
  "_source": ["dataset_id", "project_id", "name", "source", "size", "data_type"],
  "query": { "term": { "workspace_id": "demo" } }
}

# 5. Full-text search. The backend uses multi_match for project search.
GET projects/_search
{
  "size": 5,
  "_source": ["project_id", "name", "description"],
  "query": {
    "bool": {
      "filter": [
        { "term": { "workspace_id": "demo" } },
        { "multi_match": { "query": "vehicle", "fields": ["name^3", "description"], "type": "bool_prefix" } }
      ]
    }
  }
}

# 6. Exact-match status plus accuracy range, as on the Training Runs page.
GET training_runs/_search
{
  "size": 10,
  "_source": ["run_name", "model_name", "status", "metrics.accuracy"],
  "query": {
    "bool": {
      "filter": [
        { "term": { "workspace_id": "demo" } },
        { "term": { "status": "Completed" } },
        { "range": { "metrics.accuracy": { "gte": 0.9 } } }
      ]
    }
  },
  "sort": [{ "metrics.accuracy": "desc" }]
}

# 7. Aggregations power the dashboard and analytics charts.
GET training_runs/_search
{
  "size": 0,
  "query": { "term": { "workspace_id": "demo" } },
  "aggs": {
    "statuses": { "terms": { "field": "status" } },
    "average_accuracy": { "avg": { "field": "metrics.accuracy" } },
    "deployed": { "filter": { "term": { "deployment.deployed": true } } }
  }
}

# 8. A nested query finds models with a version 1.1 document.
GET models/_search
{
  "size": 10,
  "_source": ["name", "model_type", "versions.version"],
  "query": {
    "bool": {
      "filter": [
        { "term": { "workspace_id": "demo" } },
        { "nested": { "path": "versions", "query": { "term": { "versions.version": "1.1" } } } }
      ]
    }
  }
}

# OPTIONAL: Core CRUD operations directly in Elasticsearch.
# Run 9–12 in order during the demo. They create and then remove one temporary
# project document in the demo workspace. Do not run the DELETE request first.

# 9. CREATE one temporary document.
PUT projects/_doc/dbms-teacher-demo?refresh=true
{
  "project_id": "dbms-teacher-demo",
  "workspace_id": "demo",
  "name": "Teacher Demo Project",
  "description": "Created directly in Kibana Console",
  "owner_user_id": "2391fe16-35c3-5653-8a8d-3bd0ea123261",
  "created_at": "2026-10-19T00:00:00Z",
  "updated_at": "2026-10-19T00:00:00Z"
}

# 10. READ the document by its ID.
GET projects/_doc/dbms-teacher-demo

# 11. UPDATE one field.
POST projects/_update/dbms-teacher-demo?refresh=true
{
  "doc": { "description": "Updated directly in Kibana Console" }
}

# 12. DELETE the temporary document after the teacher has seen it.
DELETE projects/_doc/dbms-teacher-demo?refresh=true
